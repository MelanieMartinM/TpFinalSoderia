            try {

                ProductoRepositoryJdbc productoRepo = new ProductoRepositoryJdbc();
                MarcaRepositoryJdbc marcaRepo = new MarcaRepositoryJdbc();
                CatalogoRepositoryJdbc catalogoRepo = new CatalogoRepositoryJdbc();
                this.servicioCatalogo = new ServicioCatalogo(productoRepo, marcaRepo, catalogoRepo);
                cargarCatalogo();
            } catch ( PersistenciaException | IllegalStateException e) {
                mostrarError("Sin conexiÃ³n a BD: " + e.getMessage());
            }
        }

        private void configurarColumnas() {
            colId.setCellValueFactory(data -> new SimpleStringProperty(
                    data.getValue().getId() != null ? data.getValue().getId() : "-"
            ));
            colNombre.setCellValueFactory(data -> new SimpleStringProperty(
                    data.getValue().getNombre() != null ? data.getValue().getNombre() : "-"
            ));
            colProducto.setCellValueFactory(data -> new SimpleStringProperty(

                            ? data.getValue().getProducto().getNombre()
                            : "-"
            ));
            colPrecio.setCellValueFactory(data -> new SimpleStringProperty(
                    data.getValue().getPrecioLlenado() != null
                            ? formatoMoneda.format(data.getValue().getPrecioLlenado())
                            : formatoMoneda.format(BigDecimal.ZERO)
            ));
            colEstado.setCellValueFactory(data -> new SimpleStringProperty(
                    data.getValue().isActiva() ? "Activa" : "Inactiva"
            ));
        }

        @FXML
        public void cargarCatalogo() {
            if (servicioCatalogo == null) return;
            try {
                tblPresentaciones.setItems(FXCollections.observableArrayList(servicioCatalogo.listarPresentaciones()));
                lblEstado.setText("");
            } catch (PersistenciaException e) {
                mostrarError("Error al cargar catÃ¡logo: " + e.getMessage());
            }
        }

        @FXML
        private void onActualizarPrecio() {
            if (servicioCatalogo == null) {
                mostrarError("No hay conexiÃ³n activa con la base de datos.");
                return;

            Presentacion seleccionada = tblPresentaciones.getSelectionModel().getSelectedItem();
            if (seleccionada == null) {
                mostrarError("SeleccionÃ¡ una presentaciÃ³n de la tabla.");
                return;
            }
            try {
                String textoPrecio = txtNuevoPrecio.getText().trim().replace(',', '.');
                BigDecimal nuevoPrecio = new BigDecimal(textoPrecio);

package ar.edu.utn.tpfinalsoderia.controllers;

import ar.edu.utn.tpfinalsoderia.exceptions.PersistenciaException;
import ar.edu.utn.tpfinalsoderia.models.Cliente;
import ar.edu.utn.tpfinalsoderia.services.ServicioClientes;
import javafx.beans.property.SimpleStringProperty;
import javafx.collections.FXCollections;
import javafx.fxml.FXML;
import javafx.scene.control.Label;
import javafx.scene.control.TableColumn;
import javafx.scene.control.TableView;
import javafx.scene.control.TextField;

public class ClientesController {

    @FXML
    private TextField campoId;

    @FXML
    private TextField campoNombre;

    @FXML
    private TextField campoTelefono;

    @FXML
    private TextField campoDireccion;

    @FXML
    private Label mensaje;

    @FXML
    private TableView<Cliente> tablaClientes;

    @FXML
    private TableColumn<Cliente, String> columnaId;

    @FXML
    private TableColumn<Cliente, String> columnaNombre;

    @FXML
    private TableColumn<Cliente, String> columnaTelefono;

    @FXML
    private TableColumn<Cliente, String> columnaDireccion;

    private final ServicioClientes servicio = new ServicioClientes();

    @FXML
    public void initialize() {
        columnaId.setCellValueFactory(
                fila -> new SimpleStringProperty(fila.getValue().getId())
        );

        columnaNombre.setCellValueFactory(
                fila -> new SimpleStringProperty(fila.getValue().getNombre())
        );

        columnaTelefono.setCellValueFactory(
                fila -> new SimpleStringProperty(fila.getValue().getTelefono())
        );

        columnaDireccion.setCellValueFactory(
                fila -> new SimpleStringProperty(fila.getValue().getDireccion())
        );

        tablaClientes.getSelectionModel()
                .selectedItemProperty()
                .addListener((observable, anterior, seleccionado) -> {
                    if (seleccionado != null) {
                        mostrarEnCampos(seleccionado);
                    }
                });

        cargarClientes();
    }

    private void mostrarEnCampos(Cliente cliente) {
        campoId.setText(cliente.getId());
        campoNombre.setText(cliente.getNombre());
        campoTelefono.setText(cliente.getTelefono());
        campoDireccion.setText(cliente.getDireccion());

        // El ID no se modifica mientras se edita un cliente.
        campoId.setEditable(false);
    }

    private void cargarClientes() {
        try {
            tablaClientes.setItems(
                    FXCollections.observableArrayList(servicio.listar())
            );
        } catch (PersistenciaException | IllegalStateException e) {
            mensaje.setText("No se pudieron cargar los clientes.");
        }
    }

    @FXML
    private void guardarCliente() {
        try {
            Cliente cliente = crearClienteDesdeCampos();

            servicio.guardar(cliente);
            cargarClientes();
            limpiarCampos();

            mensaje.setText("Cliente guardado correctamente.");

        } catch (IllegalArgumentException e) {
            mensaje.setText(e.getMessage());

        } catch (PersistenciaException | IllegalStateException e) {
            mensaje.setText("No se pudo guardar el cliente.");
        }
    }

    @FXML
    private void actualizarCliente() {
        Cliente seleccionado =
                tablaClientes.getSelectionModel().getSelectedItem();

        if (seleccionado == null) {
            mensaje.setText("SeleccionÃ¡ un cliente de la tabla.");
            return;
        }

        try {
            Cliente cliente = crearClienteDesdeCampos();

            servicio.actualizar(cliente);
            cargarClientes();
            limpiarCampos();

            mensaje.setText("Cliente actualizado correctamente.");

        } catch (IllegalArgumentException e) {
            mensaje.setText(e.getMessage());

        } catch (PersistenciaException | IllegalStateException e) {
            mensaje.setText("No se pudo actualizar el cliente.");
        }
    }

    @FXML
    private void desactivarCliente() {
        Cliente seleccionado =
                tablaClientes.getSelectionModel().getSelectedItem();

        if (seleccionado == null) {
            mensaje.setText("SeleccionÃ¡ un cliente de la tabla.");
            return;
        }

        try {
            servicio.desactivar(seleccionado.getId());
            cargarClientes();
            limpiarCampos();

            mensaje.setText("Cliente desactivado correctamente.");

        } catch (PersistenciaException | IllegalStateException e) {
            mensaje.setText("No se pudo desactivar el cliente.");
        }
    }

    @FXML
    private void limpiarFormulario() {
        limpiarCampos();
        mensaje.setText("");
    }

    private void limpiarCampos() {
        tablaClientes.getSelectionModel().clearSelection();

        campoId.clear();
        campoNombre.clear();
        campoTelefono.clear();
        campoDireccion.clear();

        campoId.setEditable(true);
    }

    private Cliente crearClienteDesdeCampos() {
        Cliente cliente = new Cliente();

        cliente.setId(campoId.getText().trim());
        cliente.setNombre(campoNombre.getText().trim());
        cliente.setTelefono(campoTelefono.getText().trim());
        cliente.setDireccion(campoDireccion.getText().trim());

        return cliente;
    }
}

            Cliente titular = null;

                if (clienteId.isEmpty()) {

                }
                Presentacion presentacion = new Presentacion();
                presentacion.setId(presId);

                servicioInventario.registrarProduccion(titular, presentacion, cantidad);
                mostrarExito(lblEstado, "ProducciÃ³n de " + cantidad + " unidades registrada.");

                    txtPedidoId.getText().trim(),
                    txtDetallePedidoId.getText().trim(),
                    pres,
                    cantPedida,

        if (!chkSinCobro.isSelected()) {

                String cliId = txtClienteId.getText().trim();
                if (cliId.isEmpty()) {

import ar.edu.utn.tpfinalsoderia.exceptions.PersistenciaException;

public class CatalogoRepositoryJdbc extends RepositorioJdbc<Presentacion> {

    private static final String SQL_INSERTAR =
            "INSERT INTO presentaciones (id, nombre, producto_id, envase_id, marca_id, precio_llenado, activa) " +
                    "VALUES (?, ?, ?, ?, ?, ?, ?)";
    private static final String SQL_ACTUALIZAR =
            "UPDATE presentaciones SET nombre = ?, producto_id = ?, envase_id = ?, marca_id = ?, precio_llenado = ?, activa = ? " +
                    "WHERE id = ?";
    private static final String SQL_BAJA_LOGICA =
            "UPDATE presentaciones SET activa = FALSE WHERE id = ?";
    private static final String SQL_BUSCAR_POR_ID =
            "SELECT id, nombre, producto_id, envase_id, marca_id, precio_llenado, activa " +
                    "FROM presentaciones WHERE id = ?";
    private static final String SQL_LISTAR_ACTIVAS =
            "SELECT id, nombre, producto_id, envase_id, marca_id, precio_llenado, activa " +
                    "FROM presentaciones WHERE activa = TRUE";

    public CatalogoRepositoryJdbc() throws PersistenciaException {
        super(null);
        conectar();
    }

    private void conectar() throws PersistenciaException {
        try {
            this.conexion = ConexionFactory.abrirConexion();
        } catch (SQLException e) {
            throw new PersistenciaException("No se pudo conectar a la base de datos", e);
        }
    }

    @Override
    protected Presentacion mapear(ResultSet resultado) throws SQLException {
        Presentacion presentacion = new Presentacion();
        presentacion.setId(resultado.getString("id"));
        presentacion.setNombre(resultado.getString("nombre"));

        String productoId = resultado.getString("producto_id");
        if (productoId != null && !productoId.isBlank()) {
            Producto producto = new Producto();
            producto.setId(productoId);
            presentacion.setProducto(producto);
        }

        String envaseId = resultado.getString("envase_id");
        if (envaseId != null && !envaseId.isBlank()) {
            Envase envase = new Envase();
            envase.setId(envaseId);
            presentacion.setEnvase(envase);
        }

        String marcaId = resultado.getString("marca_id");
        if (marcaId != null && !marcaId.isBlank()) {
            Marca marca = new Marca();
            marca.setId(marcaId);
            presentacion.setMarca(marca);
        }

        BigDecimal precioLlenado = resultado.getBigDecimal("precio_llenado");
        presentacion.setPrecioLlenado(precioLlenado);
        presentacion.setActiva(resultado.getBoolean("activa"));
        return presentacion;
    }

    @Override
    public void guardar(Presentacion presentacion) throws PersistenciaException {
        try (PreparedStatement ps = conexion.prepareStatement(SQL_INSERTAR)) {
            ps.setString(1, presentacion.getId());
            ps.setString(2, presentacion.getNombre());
            setIdOrNull(ps, 3, presentacion.getProducto() != null ? presentacion.getProducto().getId() : null);
            setIdOrNull(ps, 4, presentacion.getEnvase() != null ? presentacion.getEnvase().getId() : null);
            setIdOrNull(ps, 5, presentacion.getMarca() != null ? presentacion.getMarca().getId() : null);
            ps.setBigDecimal(6, presentacion.getPrecioLlenado());
            ps.setBoolean(7, presentacion.isActiva());
            ps.executeUpdate();
        } catch (SQLException e) {
            throw new PersistenciaException("Error al guardar la presentaciÃ³n id: " + presentacion.getId(), e);
        }
    }

    @Override
    public Presentacion buscarPorId(String id) throws PersistenciaException {
        try (PreparedStatement ps = conexion.prepareStatement(SQL_BUSCAR_POR_ID)) {
            ps.setString(1, id);
            try (ResultSet rs = ps.executeQuery()) {
                if (rs.next()) {
                    return mapear(rs);
                }
                return null;
            }
        } catch (SQLException e) {
            throw new PersistenciaException("Error al buscar la presentaciÃ³n id: " + id, e);
        }
    }

    @Override
    public List<Presentacion> listar() throws PersistenciaException {
        List<Presentacion> presentaciones = new ArrayList<>();
        try (PreparedStatement ps = conexion.prepareStatement(SQL_LISTAR_ACTIVAS);
             ResultSet rs = ps.executeQuery()) {
            while (rs.next()) {
                presentaciones.add(mapear(rs));
            }
            return presentaciones;
        } catch (SQLException e) {
            throw new PersistenciaException("Error al listar las presentaciones activas", e);
        }
    }

    @Override
    public void actualizar(Presentacion presentacion) throws PersistenciaException {
        try (PreparedStatement ps = conexion.prepareStatement(SQL_ACTUALIZAR)) {
            ps.setString(1, presentacion.getNombre());
            setIdOrNull(ps, 2, presentacion.getProducto() != null ? presentacion.getProducto().getId() : null);
            setIdOrNull(ps, 3, presentacion.getEnvase() != null ? presentacion.getEnvase().getId() : null);
            setIdOrNull(ps, 4, presentacion.getMarca() != null ? presentacion.getMarca().getId() : null);
            ps.setBigDecimal(5, presentacion.getPrecioLlenado());
            ps.setBoolean(6, presentacion.isActiva());
            ps.setString(7, presentacion.getId());
            ps.executeUpdate();
        } catch (SQLException e) {
            throw new PersistenciaException("Error al actualizar la presentaciÃ³n id: " + presentacion.getId(), e);
        }
    }

    @Override
    public void eliminar(String id) throws PersistenciaException {
        try (PreparedStatement ps = conexion.prepareStatement(SQL_BAJA_LOGICA)) {
            ps.setString(1, id);
            ps.executeUpdate();
        } catch (SQLException e) {
            throw new PersistenciaException("Error al dar de baja lÃ³gica la presentaciÃ³n id: " + id, e);
        }
    }

    private void setIdOrNull(PreparedStatement ps, int index, String valor) throws SQLException {
        if (valor == null || valor.isBlank()) {
            ps.setNull(index, Types.VARCHAR);
        } else {
            ps.setString(index, valor);
        }
    }
    @Override
    public void desactivar(String id) {
        eliminar(id); // Para esta entidad, eliminar ya realiza la baja lÃ³gica.

import java.sql.Connection;

//Para que nadie tenga que subir su contraseÃ±a a GitHub,
// vamos a leer los datos de conexiÃ³n desde variables locales de cada computadora.

public final class ConexionFactory {

    private ConexionFactory() {
        // Evita crear objetos de esta clase.
    }

    public static Connection abrirConexion() throws SQLException {

        String url = System.getenv("SODERIA_DB_URL");
        String usuario = System.getenv("SODERIA_DB_USUARIO");
        String clave = System.getenv("SODERIA_DB_CLAVE");

        if (url == null || url.isBlank()
                || usuario == null || usuario.isBlank()
                || clave == null) {
            throw new IllegalStateException(
                    "Falta configurar SODERIA_DB_URL, "
                            + "SODERIA_DB_USUARIO o SODERIA_DB_CLAVE."
            );
        }

        return DriverManager.getConnection(url, usuario, clave);
    }

package ar.edu.utn.tpfinalsoderia.DAO;

import java.sql.Connection;
import java.sql.SQLException;
import java.util.Objects;

/** Gestiona una transacciÃ³n propia, usando la conexiÃ³n compartida por los DAO. */
public final class TransaccionJdbc implements AutoCloseable {
    private final Connection conexion;
    private boolean confirmada;

    public TransaccionJdbc(Connection conexion) throws SQLException {
        this.conexion = Objects.requireNonNull(conexion, "Falta la conexiÃ³n.");
        if (!conexion.getAutoCommit()) {
            throw new SQLException("La conexiÃ³n ya tiene una transacciÃ³n activa.");
        }
        conexion.setAutoCommit(false);
    }

    public void confirmar() throws SQLException {
        conexion.commit();
        confirmada = true;
    }

    @Override
    public void close() throws SQLException {
        if (!confirmada) {
            try {
                conexion.rollback();
            } catch (SQLException error) {
                // No restaurar autoCommit despuÃ©s de fallar el rollback: eso podrÃ­a confirmar datos.
                try { conexion.close(); } catch (SQLException cierre) { error.addSuppressed(cierre); }
                throw error;
            }
        }
        conexion.setAutoCommit(true);
    }
}

    public void registrarRetiroYCobro(Retiro retiro, Cobro cobro)
            throws ReglaNegocioException, StockInsuficienteException {
        if (retiro == null || retiro.getDetalles() == null || retiro.getDetalles().isEmpty()) {
            throw new ReglaNegocioException("El retiro debe tener al menos un detalle.");
        }

        Cliente titular = retiro.getCliente();

            throw new ReglaNegocioException("Debe indicar el repartidor titular.");

        for (DetalleRetiro det : retiro.getDetalles()) {
            if (det == null) {
                throw new ReglaNegocioException("Hay un detalle nulo en el retiro.");
            }
            if (det.getCantidadRetirada() <= 0) {
                throw new ReglaNegocioException("La cantidad retirada debe ser mayor a cero.");
            }
            if (det.getPrecioAplicado() == null || det.getPrecioAplicado().signum() < 0 || det.getPrecioAplicado().scale() > 2) {
                throw new ReglaNegocioException("El precio aplicado no puede ser negativo.");
            }
            if (det.getCantidadCajones() < 0)
                throw new ReglaNegocioException("La cantidad de cajones no puede ser negativa.");
            Presentacion pres = det.getPresentacion();
            if (pres == null || pres.getId() == null || pres.getId().isBlank()) {
                throw new ReglaNegocioException("Cada detalle del retiro debe indicar una presentaciÃ³n.");
            }


        }

        try (TransaccionJdbc tx = new TransaccionJdbc(conexion)) {
            java.util.Set<String> detallesProcesados = new java.util.HashSet<>();
            for (DetalleRetiro det : retiro.getDetalles()) {
                validarPedidoYPrecio(retiro, det, detallesProcesados);
            }
            BigDecimal totalRetiro = retiro.calcularTotal();
            validarCobro(cobro, totalRetiro);

            for (DetalleRetiro det : retiro.getDetalles()) {
                StockProducto stockProd = inventarioRepo.buscarStockProducto(titular, det.getPresentacion());
                int disponibles = (stockProd == null) ? 0 : stockProd.getDisponibles();

                if (disponibles < det.getCantidadRetirada()) {
                    throw new StockInsuficienteException(
                            "Stock insuficiente para '" + det.getPresentacion().getNombre()
                                    + "'. Disponible: " + disponibles + ", solicitado: " + det.getCantidadRetirada()
                    );
                }

                stockProd.setDisponibles(disponibles - det.getCantidadRetirada());
                inventarioRepo.guardarStockProducto(stockProd);

                if (det.getCantidadCajones() > 0) {
                    TipoCajon tipoCajon = det.getTipoCajon();
                    if (tipoCajon == null || tipoCajon.getId() == null || tipoCajon.getId().isBlank()) {
                        throw new ReglaNegocioException("Debe indicar el tipo de cajÃ³n cuando retira cajones.");
                    }

                    int cajonesDisp = inventarioRepo.saldoCajones(titular, tipoCajon);
                    if (cajonesDisp < det.getCantidadCajones()) {
                        throw new StockInsuficienteException(
                                "Cajones insuficientes para '" + tipoCajon.getDescripcion()
                                        + "'. Disponibles: " + cajonesDisp
                        );
                    }

                    inventarioRepo.guardarStockCajon(new StockCajon(titular, tipoCajon, cajonesDisp - det.getCantidadCajones()));
                    inventarioRepo.registrarMovimientoCajon(new MovimientoCajon(
                            UUID.randomUUID().toString(),
                            titular,
                            tipoCajon,
                            det.getCantidadCajones(),
                            LocalDateTime.now(),
                            false
                    ));
                }
            }

            if (retiro.getId() == null || retiro.getId().isBlank()) {
                retiro.setId(UUID.randomUUID().toString());
            }
            if (retiro.getFecha() == null) {
                retiro.setFecha(LocalDateTime.now());
            }

            retiroCobroRepo.guardarRetiro(retiro);

            if (cobro != null && !cobro.isSinCobroAutorizado()) {
                if (cobro.getId() == null || cobro.getId().isBlank()) {
                    cobro.setId(UUID.randomUUID().toString());
                }

                retiro.setCobro(cobro);
                retiroCobroRepo.guardarCobro(retiro.getId(), cobro);
            }
            tx.confirmar();
        } catch (SQLException e) {
            throw new PersistenciaException("Error en la transacciÃ³n de retiro y cobro.", e);
        }
    }

    private void validarPedidoYPrecio(Retiro retiro, DetalleRetiro det, java.util.Set<String> vistos)
            throws ReglaNegocioException {
        String detalleId = det.getDetallePedidoId();
        String pedidoId = retiro.getPedidoId();
        boolean tienePedido = pedidoId != null && !pedidoId.isBlank();
        boolean tieneDetalle = detalleId != null && !detalleId.isBlank();
        if (!tienePedido && !tieneDetalle) return;
        if (!tienePedido || !tieneDetalle)
            throw new ReglaNegocioException("Para retirar de un pedido se necesitan los ID del pedido y detalle.");
        if (!vistos.add(detalleId))
            throw new ReglaNegocioException("El mismo detalle de pedido aparece dos veces en el retiro.");
        String sql = "SELECT p.cliente_id, p.estado, d.presentacion_id, d.cantidad_unidades, " +
                "d.precio_unitario_pactado FROM pedidos p JOIN detalle_pedido d ON d.pedido_id=p.id " +
                "WHERE p.id=? AND d.id=? FOR UPDATE";
        try (java.sql.PreparedStatement ps = conexion.prepareStatement(sql)) {
            ps.setString(1, pedidoId); ps.setString(2, detalleId);
            try (java.sql.ResultSet rs = ps.executeQuery()) {
                if (!rs.next()) throw new ReglaNegocioException("El detalle no pertenece al pedido indicado.");
                if (!retiro.getCliente().getId().equals(rs.getString("cliente_id")))
                    throw new ReglaNegocioException("El pedido pertenece a otro repartidor.");
                String estado = rs.getString("estado");
                if (!"CONFIRMADO".equals(estado) && !"PARCIAL".equals(estado))
                    throw new ReglaNegocioException("El pedido no estÃ¡ confirmado o ya fue cerrado.");
                if (!det.getPresentacion().getId().equals(rs.getString("presentacion_id")))
                    throw new ReglaNegocioException("La presentaciÃ³n no coincide con el detalle del pedido.");
                int pedida = rs.getInt("cantidad_unidades");
                int retirada = retiroCobroRepo.obtenerCantidadYaRetirada(detalleId);
                if ((long) retirada + det.getCantidadRetirada() > pedida)
                    throw new ReglaNegocioException("La cantidad supera el saldo del pedido.");
                det.setPedidoId(pedidoId);
                det.setCantidadPedida(pedida);
                det.setPrecioAplicado(rs.getBigDecimal("precio_unitario_pactado"));
            }
        } catch (SQLException e) {
            throw new PersistenciaException("Error al verificar el detalle del pedido.", e);
        }
    }

    /** Registra un pago posterior asociado a un retiro ya guardado. */
    public void registrarCobroPosterior(String retiroId, Cobro cobro) throws ReglaNegocioException {
        if (retiroId == null || retiroId.isBlank())
            throw new ReglaNegocioException("Debe indicar el ID del retiro.");
        if (cobro == null || cobro.isSinCobroAutorizado())
            throw new ReglaNegocioException("Debe indicar un cobro efectivo.");
        validarCobro(cobro, BigDecimal.ZERO);
        try (TransaccionJdbc tx = new TransaccionJdbc(conexion)) {
            try (java.sql.PreparedStatement ps = conexion.prepareStatement(
                    "SELECT id FROM retiros WHERE id = ? FOR UPDATE")) {
                ps.setString(1, retiroId);
                try (java.sql.ResultSet rs = ps.executeQuery()) {
                    if (!rs.next()) throw new ReglaNegocioException("El retiro indicado no existe.");
                }
            }
            retiroCobroRepo.guardarCobro(retiroId, cobro);
            tx.confirmar();
        } catch (SQLException e) {
            throw new PersistenciaException("Error al registrar el pago posterior.", e);
        }
    }

    private void validarCobro(Cobro cobro, BigDecimal totalRetiro) throws ReglaNegocioException {
        // Un retiro puede quedar pendiente. El estado no implica autorizaciÃ³n adicional.
        if (cobro == null || cobro.isSinCobroAutorizado()) return;
        if (cobro.getMedioPago() == null)
            throw new ReglaNegocioException("Debe seleccionar un medio de pago.");
        if (cobro.getMonto() == null || cobro.getMonto().signum() <= 0 || cobro.getMonto().scale() > 2)
            throw new ReglaNegocioException("El monto cobrado debe ser positivo.");
    }
